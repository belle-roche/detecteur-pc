# bot_tracer.py - Sans base de données (Zero Supabase)
import os
import requests

DISCORD_WEBHOOK_URL = os.getenv("DISCORD_WEBHOOK_URL", "TON_WEBHOOK_DISCORD_ICI")

RULES = [
    # PC
    {"cat": "PC", "keywords": ["ryzen 5 5600 b550", "kit upgrade 5600"], "max_price": 220, "exclude": ["5600g", "a320", "16go"]},
    {"cat": "PC", "keywords": ["ryzen 5 5600", "ryzen 5 5600x"], "max_price": 95, "exclude": ["5600g", "pins tordus"]},
    {"cat": "PC", "keywords": ["b550", "b550m"], "max_price": 65, "exclude": ["a320", "a520", "b450"]},
    {"cat": "PC", "keywords": ["32 go ddr4", "32go ddr4"], "max_price": 55, "exclude": ["sodimm", "ecc"]},
    {"cat": "PC", "keywords": ["rx 6800"], "max_price": 275, "exclude": ["hs"]},
    {"cat": "PC", "keywords": ["rx 6800 xt", "rx 6800xt"], "max_price": 300, "exclude": ["hs"]},
    {"cat": "PC", "keywords": ["rtx 3070"], "max_price": 235, "exclude": ["hs"]},

    # VINYL
    {"cat": "VINYL", "keywords": ["vinyle u2", "vinyl u2", "joshua tree"], "max_price": 18, "exclude": ["cd", "dvd", "rayé"]},
    {"cat": "VINYL", "keywords": ["vinyle imagine dragons"], "max_price": 18, "exclude": ["cd", "dvd"]},
    {"cat": "VINYL", "keywords": ["vinyle coldplay"], "max_price": 18, "exclude": ["cd"]},
    {"cat": "VINYL", "keywords": ["vinyle dropkick murphys"], "max_price": 18, "exclude": ["cd"]},
    {"cat": "VINYL", "keywords": ["vinyle meteora", "vinyle hybrid theory", "vinyle numb"], "max_price": 22, "exclude": ["cd"]}
]

def send_discord_alert(title, price, url, source):
    payload = {
        "embeds": [{
            "title": f"🚨 NOUVELLE AFFAIRE : {title}",
            "description": f"**Prix :** {price} €\n**Plateforme :** {source}\n[Ouvrir l'annonce]({url})",
            "color": 3447003
        }]
    }
    requests.post(DISCORD_WEBHOOK_URL, json=payload)

def filter_ad(title, price):
    title_lower = title.lower()
    for rule in RULES:
        if any(kw in title_lower for kw in rule["keywords"]):
            if price <= rule["max_price"]:
                if not any(ex in title_lower for ex in rule["exclude"]):
                    return True
    return False

if __name__ == "__main__":
    print("Bot démarré...")
