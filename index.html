def evaluate_ad(title, price, shipping_available=True):
    title_lower = title.lower()
    
    # Check exclusions globales
    for excl in CONFIG["exclusions_globales"]:
        if excl in title_lower:
            return {"status": "EXCLUDED", "reason": f"Exclusion globale: '{excl}'"}
            
    if not shipping_available:
        return {"status": "EXCLUDED", "reason": "Livraison non disponible"}

    # Test contre chaque target
    for target in CONFIG["targets"]:
        kw_match = any(kw in title_lower for kw in target["keywords"])
        if kw_match:
            # Check exclusions spécifiques
            excl_match = any(excl in title_lower for excl in target["exclusions"])
            if excl_match:
                continue # Passer à une autre catégorie si exclus
                
            if price <= target["prix_jackpot"]:
                return {"status": "JACKPOT", "target": target["category"], "price": price, "max": target["prix_max"]}
            elif price <= target["prix_max"]:
                return {"status": "ALERT", "target": target["category"], "price": price, "max": target["prix_max"]}
            else:
                return {"status": "TOO_EXPENSIVE", "reason": f"Prix {price}€ > Max {target['prix_max']}€"}

    return {"status": "NO_MATCH", "reason": "Aucun mot-clé valide identifié"}

def notify_discord(ad, result):
    if not DISCORD_WEBHOOK_URL:
        return
    color = 0xf59e0b if result["status"] == "JACKPOT" else 0x10b981
    payload = {
        "embeds": [{
            "title": f"🚨 PRIX {result['status']} : {ad['title']}",
            "description": f"**Prix :** {ad['price']} € (Max: {result['max']} €)\n**Catégorie :** {result['target']}\n**Plateforme :** {ad['platform']}",
            "url": ad.get("url", "#"),
            "color": color
        }]
    }
    requests.post(DISCORD_WEBHOOK_URL, json=payload)
